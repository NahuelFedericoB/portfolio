import { userUrl } from '$lib/constants/apiUrl';

import { normalizeUsers } from '../normalizers/normalizeUsers';
import type {
  UserFilters,
  UsersResponse,
  UsersResult,
} from '../types/Users.types';

export const fetchUsers = async (
  filters: Partial<UserFilters> = {},
): Promise<UsersResult> => {
  try {
    const urlWithParams = craftQueryParams(userUrl, filters);
    const response = await fetch(urlWithParams);

    const data: UsersResponse = await response.json();

    return {
      list: normalizeUsers(data),
      length: data?.resourceCount || 0,
    };
  } catch (error) {
    return { list: [], length: 0 };
  }
};
